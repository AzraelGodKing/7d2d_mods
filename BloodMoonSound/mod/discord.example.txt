# Blood Moon Discord settings (SERVER ONLY - players can ignore this file).
#
# On a dedicated server, Blood Moon Sound can post blood moon updates to a Discord channel.
# To turn it on:
#   1. In Discord: Channel settings > Integrations > Webhooks > New Webhook > Copy Webhook URL
#   2. On the SERVER, copy this file to  discord.txt  (same folder) and paste the URL below
#   3. Restart the server, or type  bmdiscord reload  in the server console
#   4. Type  bmdiscord  to see the schedule, and  bmdiscord test  to check posting works
#
# Keep the webhook URL private: anyone who has it can post in that channel.
# Don't put discord.txt in a mod zip you give to players.
#
# In every message you can use:
#   {day}      the blood moon's day number
#   {timeleft} time left in game time, like "1 day 6 hours", "6 hours" or "45 minutes":
#              until the horde starts, or until dawn once it has (start / update messages)
#   {realtime} the same in real time, like "1h 15m" (from the server's day length)
#   {hours}    the same in whole in-game hours, like "30"
#   {days}     calendar days left until blood moon day (0 on the day itself)
#   {starts}   the in-game time the horde starts, like "22:00"
#   {dawn}     the in-game time it ends, like "04:00"
#   {players}  names of who's online       {count}  how many are online
#   {player}   who joined or left (join/leave messages)
#   {recap}    the horde recap (dawn message)
#   \n         a new line
# Discord formatting works too: **bold**, *italics*, emoji.
#
# The FIRST LINE of a message is its headline: the embed title in Discord, and the bright red
# line in game chat. Put the details after a \n so they show smaller underneath.
# Messages still word for word as in the 1.3.0 example file get the new wording automatically.

Webhook=

# Name the messages are posted under.
BotName=Blood Moon


# ---- REMINDERS before the blood moon --------------------------------------------------
# Add as many as you like, one per line:
#
#   Reminder = <days before> <hour> | <message>
#
#   days before:  0 = on blood moon day,  1 = the day before,  3 = three days before, ...
#   hour:         in-game hour, 0-23 (the horde comes at 22:00 on a normal server)
#
# Delete a line to stop that reminder. To have no reminders at all, write:  Reminder=none
# Examples you could add:
#   Reminder = 3 8  | ⚠️ **Blood Moon in {timeleft}** · Day {day}\nHorde at {starts}. Start stocking up!
#   Reminder = 0 21 | 🚨 **Blood Moon in {timeleft}!** · Day {day}\nGet inside!

Reminder = 1 12 | ⚠️ **Blood Moon tomorrow** · Day {day}\nHorde in **{timeleft}** at {starts} (about {realtime} real time). Get your base ready!
Reminder = 0 18 | 🔴 **Blood Moon tonight** · Day {day}\nHorde in **{timeleft}** at {starts} (about {realtime} real time). Get inside before dark!


# ---- DAILY COUNTDOWN (off unless you turn it on) --------------------------------------
# One post every in-game day at CountdownHour, starting CountdownFromDays days before.
# (It stops the day before blood moon day; use a Reminder for the day itself.)
Countdown=off
CountdownHour=8
CountdownFromDays=7
MsgCountdown=⏳ **Blood Moon in {timeleft}** · Day {day}\nHorde at {starts} (about {realtime} real time).


# ---- DURING THE BLOOD MOON --------------------------------------------------------------
# When the horde starts:
Start=on
MsgStart=🌕 **The Blood Moon has risen!** · Day {day}\nDawn in {timeleft} at {dawn}. Online: {players}

# Extra posts while the horde is on, every this many in-game hours (0 = off).
UpdateEveryHours=0
MsgUpdate=🩸 **The Blood Moon rages on** · Day {day}\nDawn in {timeleft} at {dawn}. Still standing: {players}

# At dawn afterwards:
End=on
MsgEnd=🌅 **Dawn! Blood Moon survived** · Day {day}\nOnline: {players}

# Add a horde recap to the dawn post: zombies killed, kills per player, and who died.
# Put {recap} in MsgEnd to choose where it goes; otherwise it's added at the end.
Recap=on


# ---- SERVER EVENTS (off unless you turn them on) ----------------------------------------
# {player} is who joined or left; {count} and {players} are who's online now.
ServerStart=off
MsgServerStart=🟢 The server is up.
ServerStop=off
MsgServerStop=🔴 The server is shutting down.
PlayerJoin=off
MsgJoin=➡️ **{player}** joined. Online: {count}
PlayerLeave=off
MsgLeave=⬅️ **{player}** left. Online: {count}


# ---- IN-GAME CHAT ----------------------------------------------------------------------
# Also show messages in the game's chat, so players without Discord see them.
# Works even with no webhook. Emoji and **bold** are left out in game chat.
# InGameChatFor: any of  reminder, countdown, start, update, end, serverstart, serverstop,
#                join, leave  (comma separated), or  all
InGameChat=on
InGameChatFor=reminder, countdown, start, end


# ---- OTHER ------------------------------------------------------------------------------
# Skip the start / during / dawn posts when nobody is online.
SkipEmptyServer=on

# Coloured Discord embeds (red for the horde, orange for reminders, gold at dawn...).
# off = plain text messages, as before 1.3.0.
Embeds=on

# Allow @everyone / @role pings written in the messages above (off = never ping anyone).
Mentions=off
# A message that pings is sent as plain text (Discord doesn't ping from an embed).
